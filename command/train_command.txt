#!/usr/bin/env bash
set -u
TRAIN_SCRIPT='model_training_sumo_v1.py'
DEFAULT_NET_FILE='data/net/test_random_offsets_seed20260708.net.xml'

TIME_LIMIT=7200
TEST_SIZE=0.2
RANDOM_STATE=42

run_training() {
    GROUP_NAME="$1"
    NET_FILE="${2:-${DEFAULT_NET_FILE}}"
    OUTPUT_NAME="${3:-${GROUP_NAME}_no_change}"
    CSV_FILE="data/traci/TraCI_output_adjusted_${OUTPUT_NAME}.csv"
    MODEL_PATH="models/${OUTPUT_NAME}"
    LOG_FILE="training_logs/train_${OUTPUT_NAME}.log"

    echo "=================================================="
    echo "Starting training: ${GROUP_NAME}"
    echo "CSV: ${CSV_FILE}"
    echo "Model: ${MODEL_PATH}"
    echo "Log: ${LOG_FILE}"
    echo "Start time: $(date '+%Y-%m-%d %H:%M:%S')"
    echo "=================================================="

    if python3 "${TRAIN_SCRIPT}" \
        --csv "${CSV_FILE}" \
        --net "${NET_FILE}" \
        --save_path "${MODEL_PATH}" \
        --time_limit "${TIME_LIMIT}" \
        --test_size "${TEST_SIZE}" \
        --random_state "${RANDOM_STATE}" \
        > "${LOG_FILE}" 2>&1
    then
        echo "[SUCCESS] ${GROUP_NAME} completed at $(date '+%Y-%m-%d %H:%M:%S')"
    else
        EXIT_CODE=$?
        echo "[FAILED] ${GROUP_NAME}, exit code: ${EXIT_CODE}"
        echo "[FAILED] Check log: ${LOG_FILE}"
    fi

    echo
}

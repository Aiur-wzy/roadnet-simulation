# 1. free_all
nohup python3 TraCI_Python_Adjusted.py \
  --sumo-config data/sumocfg/free_all_no_change.sumocfg \
  --net-file data/net/test_random_offsets_seed20260708.net.xml \
  --sumo-binary sumo \
  --outputs legacy \
  --legacy-edge-output data/traci/TraCI_output_adjusted_free_all_no_change.csv \
  --stop-speed-threshold 0.1 \
  --low-speed-threshold 2.0 \
  --downstream-block-occupancy-threshold 85 \
  > data/traci/traci_free_all_no_change.log 2>&1 &

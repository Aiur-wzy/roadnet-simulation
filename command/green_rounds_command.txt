# Collect whole-approach missed-green rounds for datasets 12-14.
mkdir -p data/traci

# 12. free
python3 scripts/collect_vehicle_green_rounds.py \
  --sumo-config data/sumocfg/free.sumocfg \
  --sumo-binary sumo \
  --waiting-region-length 600 \
  --tripinfo-output data/tripinfo/tripinfo_free.xml \
  --vehicle-output data/traci/TraCI_vehicle_green_rounds_free.csv \
  --event-output data/traci/TraCI_vehicle_movement_green_rounds_free.csv \
  --distribution-output data/traci/TraCI_green_rounds_distribution_free.csv \
  --summary-output data/traci/TraCI_green_rounds_summary_free.txt
